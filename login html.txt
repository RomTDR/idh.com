  <script>
    // Gestion des onglets
    const tabLogin = document.getElementById('tabLoginBtn');
    const tabRegister = document.getElementById('tabRegisterBtn');
    const loginForm = document.getElementById('loginForm');
    const registerForm = document.getElementById('registerForm');

    tabLogin.addEventListener('click', () => {
      tabLogin.classList.add('active');
      tabLogin.style.background = '#004225';
      tabLogin.style.color = 'white';
      tabRegister.classList.remove('active');
      tabRegister.style.background = '#f0f0f0';
      tabRegister.style.color = '#333';
      loginForm.style.display = 'block';
      registerForm.style.display = 'none';
    });

    tabRegister.addEventListener('click', () => {
      tabRegister.classList.add('active');
      tabRegister.style.background = '#004225';
      tabRegister.style.color = 'white';
      tabLogin.classList.remove('active');
      tabLogin.style.background = '#f0f0f0';
      tabLogin.style.color = '#333';
      loginForm.style.display = 'none';
      registerForm.style.display = 'block';
    });

    // Vérifier si déjà connecté
    async function checkAlreadyLogged() {
      const { data: { session } } = await supabaseClient.auth.getSession();
      if (session) {
        window.location.href = 'membres.html';
      }
    }
    checkAlreadyLogged();

    // Gestion connexion
    async function handleLogin(e) {
      e.preventDefault();
      const email = document.getElementById('loginEmail').value;
      const password = document.getElementById('loginPassword').value;
      
      const { error } = await IDH.signIn(email, password);
      
      if (error) {
        document.getElementById('loginAlert').innerHTML = `<div class="alert alert-error">${error.message}</div>`;
      } else {
        window.location.href = 'membres.html';
      }
    }

    // Gestion inscription
    async function handleRegister(e) {
      e.preventDefault();
      
      const pwd1 = document.getElementById('regPassword').value;
      const pwd2 = document.getElementById('regPassword2').value;
      
      if (pwd1 !== pwd2) {
        document.getElementById('registerAlert').innerHTML = '<div class="alert alert-error">Les mots de passe ne correspondent pas.</div>';
        return;
      }
      
      const email = document.getElementById('regEmail').value;
      const password = pwd1;
      const firstName = document.getElementById('regFirstName').value;
      const lastName = document.getElementById('regLastName').value;
      const specialty = document.getElementById('regSpecialty').value;
      const hospital = document.getElementById('regHospital').value;
      const year = document.getElementById('regYear').value;
      const phone = document.getElementById('regPhone').value;
      
      const { error } = await IDH.signUp(email, password, { 
        first_name: firstName, 
        last_name: lastName,
        specialty: specialty,
        hospital: hospital,
        promotion_year: year,
        phone: phone
      });
      
      if (error) {
        document.getElementById('registerAlert').innerHTML = `<div class="alert alert-error">${error.message}</div>`;
      } else {
        document.getElementById('registerAlert').innerHTML = '<div class="alert alert-success">✅ Inscription réussie ! Vérifiez votre email pour confirmer votre compte.</div>';
        document.getElementById('registerForm').reset();
        setTimeout(() => {
          tabLogin.click();
        }, 3000);
      }
    }

    // Mot de passe oublié
    async function forgotPassword(e) {
      e.preventDefault();
      const email = document.getElementById('loginEmail').value;
      if (!email) {
        document.getElementById('loginAlert').innerHTML = '<div class="alert alert-error">Veuillez entrer votre email.</div>';
        return;
      }
      const { error } = await supabaseClient.auth.resetPasswordForEmail(email, {
        redirectTo: window.location.origin + '/login.html'
      });
      if (error) {
        document.getElementById('loginAlert').innerHTML = `<div class="alert alert-error">${error.message}</div>`;
      } else {
        document.getElementById('loginAlert').innerHTML = '<div class="alert alert-success">✅ Email de réinitialisation envoyé !</div>';
      }
    }

    window.handleLogin = handleLogin;
    window.handleRegister = handleRegister;
    window.forgotPassword = forgotPassword;
  </script>
